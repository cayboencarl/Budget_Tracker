"Migrations are like "version control for your database schema."

Step 1: Install Flask-Migrate
<pip install flask-migrate>

Step 2: Set up migrations in your app.py
Import Migrate from flask_migrate <from flask_migrate import Migrate>
Create a Migrate object that connects Flask, SQLAlchemy, and the migration system <migrate = Migrate(app, db)>
Initialize it with your app and db <(app, db)>

Step 3: Initialize the migration folder
From your terminal (in your backend folder), run:
<flask db init>
This creates a migrations folder that will track all your schema changes.

Step 4: Create an initial migration
Run:
<flask db migrate -m "Initial migration: create transactions table">
This creates a migration file that describes your current schema.

Step 5: Apply the migration
Run:
<flask db upgrade>
This applies the migration (though the table already exists from db.create_all()).

#Migration Commands
<flask db init> # initialize the migration system (do this once)
<flask db migrate> # create a new migration based on model changes
<flask db upgrade> # apply migrations to the database
<flask db downgrade> # undo the last migration

/**
 * ========================================
 * BUDGET TRACKER - FRONTEND SCRIPT
 * ========================================
 * This script handles all frontend functionality:
 * - Fetching transactions from the API
 * - Creating new transactions
 * - Displaying transactions on the page
 */

/**
 * Fetch all transactions from the API
 * Makes a GET request to /transactions endpoint
 * Then displays them on the page
 */
async function fetchTransactions() {
    try {
        // Send GET request to the API
        const response = await fetch('http://localhost:5000/transactions');
        
        // Convert the response to JSON
        const data = await response.json();
        
        // Display the transactions on the page
        displayTransactions(data);
    } catch (error) {
        console.error('Error fetching transactions:', error);
    }
}

/**
 * Display transactions on the page
 * Creates a div for each transaction and adds it to the DOM
 * 
 * @param {Array} transactions - Array of transaction objects from the API
 */
function displayTransactions(transactions) {
    // Get the container where transactions will be displayed
    const container = document.getElementById('transactionsList');
    
    // Clear previous content
    container.innerHTML = '';
    
    // Loop through each transaction
    transactions.forEach(transaction => {
        // Create a new div element for this transaction
        const div = document.createElement('div');
        
        // Set the text content (type - amount - category - description)
        div.textContent = `${transaction.type} - $${transaction.amount} - ${transaction.category} - ${transaction.description}`;
        
        // Add the div to the container
        container.appendChild(div);
    });
}

/**
 * Create a new transaction from the form
 * Gets form values, sends POST request to API, then refreshes the list
 * 
 * @param {Event} event - The form submission event
 */
async function createTransaction(event) {
    // Prevent the form from refreshing the page
    event.preventDefault();

    // Get the values from each form input
    const type = document.getElementById('type').value;
    const amount = document.getElementById('amount').value;
    const category = document.getElementById('category').value;
    const description = document.getElementById('description').value;
    const date = document.getElementById('date').value;
    
    // Create an object with the transaction data
    const transactionData = {
        type: type,
        amount: parseFloat(amount),  // Convert to number
        category: category,
        description: description,
        date: date
    };

    try {
        // Send POST request to create the transaction
        const response = await fetch('http://localhost:5000/transactions', {
            method: 'POST',  // HTTP method for creating
            headers: {
                'Content-Type': 'application/json'  // Tell server we're sending JSON
            },
            body: JSON.stringify(transactionData)  // Convert object to JSON string
        });
        
        // Get the response from the server
        const result = await response.json();
        
        // Log the result for debugging
        console.log('Transaction created:', result);
        
        // Refresh the transaction list to show the new transaction
        fetchTransactions();
        
        // Clear the form inputs
        document.getElementById('transactionForm').reset();
    } catch (error) {
        console.error('Error creating transaction:', error);
    }
}

/**
 * ========================================
 * EVENT LISTENERS - Connect buttons/forms to functions

// When the "Fetch Transactions" button is clicked, fetch and display transactions
document.getElementById('fetchBtn').addEventListener('click', fetchTransactions);

// When the form is submitted, create a new transaction
document.getElementById('transactionForm').addEventListener('submit', createTransaction);

async function handleLogin(event){
    event.preventDefault();

    const username = document.getElementById("loginUsername").value;
    const password = document.getElementById("loginPassword").value;

    try{
        const response = await fetch("http://localhost:5000/login",{
            method: "POST",
            headers: {
                "Content-Type": "application/json"
            },
            body:JSON.stringify({username, password})
        });
        const result = await response.json();

        if (response.ok){
            console.log("Logged in", result);
            document.getElementById("loginSection").style.display = "none";
            document.getElementById("registerSection").style.display = "none";
            document.getElementById("appSection").style.display = "block";

            document.getElementById("loginForm").reset();
        }else {
            alert("Login failed: " + result.error);
        }
    }catch(error){
        console.error("Error logging in: ", error);
        alert("Error: " + error);
    }
}

async function handleRegister(event) {

    }
}
